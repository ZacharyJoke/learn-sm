# 练习 07 · SM2 数字签名服务接口

把 SM2 签名/验签能力包装成一个最小 Spring Boot REST 服务：客户端 POST 原文得到签名 Hex，再带消息与签名请求验签；错误签名返回 400。并用 JUnit 5 + MockMvc 完成自动化测试。

---

## 🎯 练习目标

- 搭建最小 Spring Boot Web 工程（`spring-boot-starter-web`）
- 实现 `POST /api/sign`：请求体即原文（`text/plain`），返回签名 Hex
- 实现 `POST /api/verify`：入参 `keyId + message + signature`，合法返回 200，签名不符返回 400
- 私钥从环境变量（`SIGN_KEY_HEX`）或配置文件加载，代码零硬编码
- 使用 JUnit 5 + MockMvc 编写接口测试，覆盖「签名→验签成功」与「篡改→400」两条路径

### 步骤 6：核心业务类 `SignatureService.java`

文件路径：`~/gm-practices/gm-sign-service/src/main/java/com/example/sign/SignatureService.java`

```java
package com.example.sign;

import org.bouncycastle.jcajce.provider.asymmetric.ec.BCECPrivateKey;
import org.bouncycastle.jcajce.provider.asymmetric.ec.BCECPublicKey;
import org.bouncycastle.jcajce.spec.SM2ParameterSpec;
import org.bouncycastle.jce.provider.BouncyCastleProvider;
import org.bouncycastle.jce.ECNamedCurveTable;
import org.bouncycastle.jce.spec.ECNamedCurveParameterSpec;
import org.bouncycastle.jce.spec.ECPrivateKeySpec;
import org.bouncycastle.jce.spec.ECPublicKeySpec;
import org.bouncycastle.math.ec.ECPoint;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import java.math.BigInteger;
import java.nio.charset.StandardCharsets;
import java.security.KeyFactory;
import java.security.KeyPair;
import java.security.KeyPairGenerator;
import java.security.PrivateKey;
import java.security.PublicKey;
import java.security.SecureRandom;
import java.security.Security;
import java.security.Signature;
import java.security.spec.ECGenParameterSpec;
import java.util.HexFormat;
import java.util.LinkedHashMap;
import java.util.Map;

@Service
public class SignatureService {

    private static final Logger log = LoggerFactory.getLogger(SignatureService.class);
    private static final HexFormat HEX = HexFormat.of();
    private static final ECNamedCurveParameterSpec CURVE =
            ECNamedCurveTable.getParameterSpec("sm2p256v1");

    // 在静态块注册 BC：测试环境不走 main()，也能保证 "BC" 可用
    static {
        Security.addProvider(new BouncyCastleProvider());
    }

    private final byte[] userId;
    private final SecureRandom random = new SecureRandom();
    private final Map<String, KeyPair> keyPairs = new LinkedHashMap<>();

    public SignatureService(SignProperties properties) throws Exception {
        String id = properties.userId() == null ? "1234567812345678" : properties.userId();
        this.userId = id.getBytes(StandardCharsets.UTF_8);

        String envKeyHex = System.getenv("SIGN_KEY_HEX");
        if (envKeyHex != null && !envKeyHex.isBlank()) {
            keyPairs.put("k1", buildFromPrivateHex(envKeyHex.strip()));
            log.info("Loaded signing key 'k1' from environment variable SIGN_KEY_HEX");
        } else if (properties.keys() != null) {
            for (Map.Entry<String, SignProperties.KeyMaterial> entry : properties.keys().entrySet()) {
                keyPairs.put(entry.getKey(), buildKeyPair(entry.getValue()));
            }
        }
        if (keyPairs.isEmpty()) {
            log.warn("No signing key configured (SIGN_KEY_HEX / app.sign.keys); "
                    + "generating an EPHEMERAL key pair for dev only");
            KeyPairGenerator keyPairGenerator = KeyPairGenerator.getInstance("EC", "BC");
            keyPairGenerator.initialize(new ECGenParameterSpec("sm2p256v1"), random);
            KeyPair keyPair = keyPairGenerator.generateKeyPair();
            keyPairs.put("k1", keyPair);
            BCECPublicKey publicKey = (BCECPublicKey) keyPair.getPublic();
            log.warn("Ephemeral public key (04||X||Y hex): {}",
                    HEX.formatHex(publicKey.getQ().getEncoded(false)));
        }
    }

    public String sign(String keyId, String message) throws Exception {
        KeyPair keyPair = keyPairs.get(keyId);
        if (keyPair == null) {
            throw new IllegalArgumentException("unknown keyId: " + keyId);
        }
        Signature signature = Signature.getInstance("SM3withSM2", "BC");
        signature.setParameter(new SM2ParameterSpec(userId));
        signature.initSign(keyPair.getPrivate(), random);
        signature.update(message.getBytes(StandardCharsets.UTF_8));
        return HEX.formatHex(signature.sign());
    }

    public boolean verify(String keyId, String message, String signatureHex) throws Exception {
        KeyPair keyPair = keyPairs.get(keyId);
        if (keyPair == null) {
            throw new IllegalArgumentException("unknown keyId: " + keyId);
        }
        Signature signature = Signature.getInstance("SM3withSM2", "BC");
        signature.setParameter(new SM2ParameterSpec(userId));
        signature.initVerify(keyPair.getPublic());
        signature.update(message.getBytes(StandardCharsets.UTF_8));
        return signature.verify(HEX.parseHex(signatureHex));
    }

    private KeyPair buildKeyPair(SignProperties.KeyMaterial material) throws Exception {
        KeyFactory keyFactory = KeyFactory.getInstance("EC", "BC");
        PrivateKey privateKey = keyFactory.generatePrivate(
                new ECPrivateKeySpec(new BigInteger(1, HEX.parseHex(material.privateHex())), CURVE));
        PublicKey publicKey;
        if (material.publicHex() != null && !material.publicHex().isBlank()) {
            ECPoint point = CURVE.getCurve().decodePoint(HEX.parseHex(material.publicHex()));
            publicKey = keyFactory.generatePublic(new ECPublicKeySpec(point, CURVE));
        } else {
            BCECPrivateKey bcPrivateKey = (BCECPrivateKey) privateKey;
            ECPoint point = CURVE.getG().multiply(bcPrivateKey.getD()).normalize();
            publicKey = keyFactory.generatePublic(new ECPublicKeySpec(point, CURVE));
        }
        return new KeyPair(publicKey, privateKey);
    }

    private KeyPair buildFromPrivateHex(String privateHex) throws Exception {
        return buildKeyPair(new SignProperties.KeyMaterial(privateHex, null));
    }
}
```

### 步骤 8：控制器 `SignController.java`

文件路径：`~/gm-practices/gm-sign-service/src/main/java/com/example/sign/SignController.java`

```java
package com.example.sign;

import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/api")
public class SignController {

    private static final String DEFAULT_KEY_ID = "k1";

    private final SignatureService signatureService;

    public SignController(SignatureService signatureService) {
        this.signatureService = signatureService;
    }

    @PostMapping(path = "/sign", consumes = MediaType.TEXT_PLAIN_VALUE)
    public SignDtos.SignResponse sign(@RequestBody String message) throws Exception {
        String signature = signatureService.sign(DEFAULT_KEY_ID, message);
        return new SignDtos.SignResponse(DEFAULT_KEY_ID, signature);
    }

    @PostMapping("/verify")
    public ResponseEntity<SignDtos.VerifyResponse> verify(
            @RequestBody SignDtos.VerifyRequest request) throws Exception {
        boolean valid = signatureService.verify(request.keyId(), request.message(), request.signature());
        SignDtos.VerifyResponse body = new SignDtos.VerifyResponse(request.keyId(), valid);
        return valid ? ResponseEntity.ok(body) : ResponseEntity.badRequest().body(body);
    }
}
```

### 步骤 9：测试类 `SignControllerTest.java`

文件路径：`~/gm-practices/gm-sign-service/src/test/java/com/example/sign/SignControllerTest.java`

```java
package com.example.sign;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.AutoConfigureMockMvc;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.http.MediaType;
import org.springframework.test.context.DynamicPropertyRegistry;
import org.springframework.test.context.DynamicPropertySource;
import org.springframework.test.web.servlet.MockMvc;

import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
class SignControllerTest {

    @Autowired
    MockMvc mockMvc;

    @Autowired
    ObjectMapper objectMapper;

    // 一次性测试密钥（仅用于本测试，生产环境必须用 KMS/环境注入，严禁照搬到生产）
    @DynamicPropertySource
    static void registerKey(DynamicPropertyRegistry registry) {
        registry.add("app.sign.keys.k1.private-hex",
                () -> "fb5dbcfeb013ac98015a89d82e771d07a4a620566e2a44e15d4f48369171f433");
        registry.add("app.sign.keys.k1.public-hex",
                () -> "04b66aa03134d6c3f46b3917c93533ad299a6476073ffc91882c5c4a6e121ebb1cbce075667b54c575b8043389c18a3c44c5c6920fc329df58bcae5dd966466ff0");
    }

    @Test
    void signThenVerifyOk() throws Exception {
        String message = "订单金额 100 元";

        String signResponse = mockMvc.perform(post("/api/sign")
                        .contentType(MediaType.TEXT_PLAIN)
                        .content(message))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.keyId").value("k1"))
                .andExpect(jsonPath("$.signature").isNotEmpty())
                .andReturn().getResponse().getContentAsString();

        JsonNode signJson = objectMapper.readTree(signResponse);
        String signature = signJson.get("signature").asText();

        String verifyBody = objectMapper.writeValueAsString(
                new SignDtos.VerifyRequest("k1", message, signature));
        mockMvc.perform(post("/api/verify")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content(verifyBody))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.valid").value(true));
    }

    @Test
    void verifyTamperedMessageReturns400() throws Exception {
        String signResponse = mockMvc.perform(post("/api/sign")
                        .contentType(MediaType.TEXT_PLAIN)
                        .content("原始消息"))
                .andReturn().getResponse().getContentAsString();
        String signature = objectMapper.readTree(signResponse).get("signature").asText();

        String verifyBody = objectMapper.writeValueAsString(
                new SignDtos.VerifyRequest("k1", "被篡改的消息", signature));
        mockMvc.perform(post("/api/verify")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content(verifyBody))
                .andExpect(status().isBadRequest())
                .andExpect(jsonPath("$.valid").value(false));
    }
}
```

## 🧯 常见失败与排查

| 现象 | 原因 | 解决 |
|------|------|------|
| 测试报 `no such provider: BC` | Provider 只在 `main()` 注册，测试不走 main | 使用 `SignatureService` 中的 static 块注册 |
| curl 中文变成乱码 | 没告诉服务端字符集，或没用 `--data-binary` | 加 `Content-Type: text/plain; charset=utf-8`，用 `--data-binary` |
| 签名每次不同导致困惑 | SM2 签名含随机数 k | 这是正确的，验签不依赖 k |
| `unknown keyId` | verify 请求的 keyId 与加载的不一致 | 用 `k1`，多密钥时先确认加载日志 |
| 服务重启后旧签名验不过 | 用了临时密钥（未配置任何 key） | 配置 `SIGN_KEY_HEX` 或 `app.sign.keys` |
| 端口 8080 被占用 | 本机其他进程占用 | 修改 `server.port` 或停掉占用进程 |
| Windows 下环境变量不生效 | 注入语法不同 | 用 `set SIGN_KEY_HEX=...` 或 IDE 运行配置 |

> API 版本提示：本文基于 Spring Boot 3.4.6 + bcprov-jdk18on 1.84、JDK 17/21 实测。版本升级时以对应 Javadoc 为准；`SM2ParameterSpec` 包名为 `org.bouncycastle.jcajce.spec`。
